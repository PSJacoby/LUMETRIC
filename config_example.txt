CAMERA_CONFIG_GROUP = Camera

# Acquisition Mode
# true = Use circular buffer (REQUIRES HARDWARE TRIGGERS from Arduino/external source)

# IMPORTANT: Demo camera will NOT work with circular buffer mode!
USE_CIRCULAR_BUFFER = false

# Each preset name becomes a channel name; its State value (2,4,8,...) maps to PIN_2, PIN_3, PIN_4,...
